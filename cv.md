## Code example

["RGB to HEX conversion": task description](https://www.codewars.com/kata/513e08acc600c94f01000001)

```js
function componentToHex(number) {
  const MIN_VALUE = 0;
  const MAX_VALUE = 255;

  return Math.max(MIN_VALUE, Math.min(MAX_VALUE, number))
    .toString(16)
    .toUpperCase()
    .padStart(2, '0');
}

function rgb(r, g, b) {
  const colors = [r, g, b];
  return colors.map((color) => componentToHex(color)).join('');
}
```
